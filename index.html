function renderMatrixWithData(rows, workingHours) { if (!rows || rows.length <= 1) return 0; const headerRow = rows[0]; let dates = []; let dateStartIndex = 2; // ค้นหาคอลัมน์เริ่มต้นของวันที่ for (let c = 0; c < headerRow.length; c++) { let cellVal = String(headerRow[c] || "").trim(); if (cellVal.includes('/') || (typeof headerRow[c] === 'number' && headerRow[c] > 30000)) { dateStartIndex = c; break; } } for (let c = dateStartIndex; c < headerRow.length; c++) { if (headerRow[c] !== undefined && headerRow[c] !== null && String(headerRow[c]).trim() !== "") { dates.push(formatExcelDate(headerRow[c])); } } if (dates.length > 0) { currentDates = dates; initMatrixTable(currentDates); } let updatedCount = 0; const tbody = document.getElementById('matrix-tbody'); const modelRows = tbody.querySelectorAll('tr'); // วนลูปอ่านข้อมูลแต่ละแถวใน Excel for (let i = 1; i < rows.length; i++) { const r = rows[i]; if (!r || r.length === 0) continue; let matchedModel = ""; for (let c = 0; c < dateStartIndex; c++) { let cellText = String(r[c] || "").trim(); productStructure.forEach(group => { group.models.forEach(m => { if (m.toLowerCase() === cellText.toLowerCase()) { matchedModel = m; } }); }); if (matchedModel) break; } if (!matchedModel) continue; if (!planDatabase[matchedModel]) planDatabase[matchedModel] = {}; modelRows.forEach(row => { const fixedModel = row.getAttribute('data-model'); if (fixedModel && fixedModel.toLowerCase() === matchedModel.toLowerCase()) { currentDates.forEach((dateText, idx) => { const rawTarget = parseFloat(r[dateStartIndex + idx]) || 0; const targetVal = Math.round(rawTarget); if (targetVal > 0) { planDatabase[matchedModel][dateText] = targetVal; const calculatedUph = workingHours > 0 ? Math.round(targetVal / workingHours) : 0; // เช็คความปลอดภัย ป้องกัน Error แถวหรือเซลล์ไม่ตรงกัน const cellIndex = idx + 2; if (row.cells && row.cells[cellIndex]) { row.cells[cellIndex].innerHTML = `
${targetVal.toLocaleString()}
UPH: ${calculatedUph}
`; updatedCount++; } } }); } }); } calculateTotals(); return updatedCount; }
